# ============================================================================
# Installation Instructions
# ============================================================================
#
# 1. Create conda environment:
#    conda create -n quantum python=3.11
#    conda activate quantum
#
# 2. Install PyTorch with CUDA support:
#    pip install torch torchvision torchaudio --index-url https://download.pytorch.org/whl/cu121
#
# 3. Install PennyLane with GPU support:
#    pip install pennylane pennylane-lightning pennylane-lightning-gpu
#
# 4. Install remaining requirements:
#    pip install -r requirements.txt
#
# 5. Verify GPU setup:
#    python test_gpu_setup.py
#
# ============================================================================
# Notes
# ============================================================================
# - pennylane-lightning-gpu requires NVIDIA GPU with CUDA support
# - If you don't have a CUDA GPU, remove pennylane-lightning-gpu and use
#   QUANTUM_DEVICE = 'lightning.qubit' (CPU) instead
# - Set NUM_WORKERS = 0 in training config to avoid multiprocessing issues
#   with quantum circuits
